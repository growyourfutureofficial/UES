function doPost(e) {
  // 🚀 IIT EXPERT FIX 1: Direct Run/GET Request Crash Failsafe
  if (!e || !e.postData || !e.postData.contents) {
    return ContentService.createTextOutput(JSON.stringify({
      "status": "error", 
      "message": "Secure IIT Engine: Direct execution blocked. Please use the App interface."
    })).setMimeType(ContentService.MimeType.JSON);
  }

  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var sheet = ss.getSheetByName("Credentials");
    
    // 🚀 IIT EXPERT FIX: Clean Database Headers Auto-Generator Engine
    if (!sheet) {
      sheet = ss.insertSheet("Credentials");
      sheet.appendRow([
        "Timestamp", "Full Name", "Email", "Password", "Referral Code", 
        "Daily Bonus", "Total Days", "Last Claim Date", "Total Wallet Balance (₹)", "Account Status"
      ]);
    }

    // 🚀 NEW ENGINE: "All Deposits" Sheet Setup
    var depositsSheet = ss.getSheetByName("All Deposits");
    if (!depositsSheet) {
      depositsSheet = ss.insertSheet("All Deposits");
      depositsSheet.appendRow([
        "Timestamp", "Full Name", "Email", "Plan Price (₹)", "Plan Name", 
        "System Hold", "Transaction ID", "Is Plan Active", "Screenshot Link"
      ]);
    }

    // 🚀 IIT EXPERT FIX: "All Withdraws" Sheet Setup
    var withdrawsSheet = ss.getSheetByName("All Withdraws");
    if (!withdrawsSheet) {
      withdrawsSheet = ss.insertSheet("All Withdraws");
      withdrawsSheet.appendRow([
        "Timestamp", "Full Name", "Email", "Account Number", "IFSC Code", 
        "Bank Name", "Withdraw Amount", "Status"
      ]);
    }

    // 🚀 IIT EXPERT FIX: "All Submits" Sheet Setup
    var submitsSheet = ss.getSheetByName("All Submits");
    if (!submitsSheet) {
      submitsSheet = ss.insertSheet("All Submits");
      submitsSheet.appendRow([
        "Timestamp", "Full Name", "Email", "Status", "Page Link", "Expected Profit"
      ]);
    }

    var data = JSON.parse(e.postData.contents);
    var action = data.action;

    // ==========================================
    // 0. ADMIN ENGINE AUTHENTICATION (IIT SECURED)
    // ==========================================
    if (action === 'adminLogin') {
      var inputUser = data.username.toString().trim();
      var inputPass = data.password.toString().trim();

      // 🚀 LAYER 1: BRUTE FORCE PROTECTION ENGINE (Server-Side Rate Limiting)
      var cache = CacheService.getScriptCache();
      var attemptsKey = "admin_attempts_" + inputUser;
      var failedAttempts = cache.get(attemptsKey);

      // Agar 5 se zyada failed attempts hain, toh seedha block kar do bina DB check kiye
      if (failedAttempts && parseInt(failedAttempts) >= 5) {
        return ContentService.createTextOutput(JSON.stringify({
          "status": "error", 
          "message": "Security Alert: Too many failed attempts. Try again after 15 minutes."
        })).setMimeType(ContentService.MimeType.JSON);
      }

      var adminSheet = ss.getSheetByName("admin credentials");
      if (!adminSheet) {
        adminSheet = ss.insertSheet("admin credentials");
        adminSheet.appendRow(["Username", "Password Hash"]);
        adminSheet.appendRow(["admin", "8d969eef6ecad3c29a3a629280e686cf0c3f5d5a86aff3ca12020c923adc6c92"]); 
      }

      var adminData = adminSheet.getDataRange().getValues();

      for (var a = 1; a < adminData.length; a++) {
        var dbUser = adminData[a][0].toString().trim();
        
        if (dbUser === inputUser) {
          // 🚀 LAYER 2: SECURE SHA-256 HASH VERIFICATION
          // Jo password frontend se aayega, server usko encrypt karke DB wale encrypt code se match karega
          var rawHash = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, inputPass);
          var txtHash = '';
          for (var i = 0; i < rawHash.length; i++) {
            var hashVal = rawHash[i];
            if (hashVal < 0) { hashVal += 256; }
            if (hashVal.toString(16).length == 1) { txtHash += '0'; }
            txtHash += hashVal.toString(16);
          }

          var dbHash = adminData[a][1].toString().trim();

          // Successful Login
          if (txtHash === dbHash) {
            cache.remove(attemptsKey); 
            
            // 🚀 SECURE IIT ENGINE: Store Token in Server Memory (Cache) for 6 Hours
            var sessionToken = Utilities.getUuid();
            var scriptCache = CacheService.getScriptCache();
            scriptCache.put("ADMIN_TOKEN_" + sessionToken, "VALID", 21600);
            
            return ContentService.createTextOutput(JSON.stringify({
              "status": "success", 
              "message": "Admin Authenticated",
              "adminToken": sessionToken 
            })).setMimeType(ContentService.MimeType.JSON);
          }
        }
      }

      // 🚀 FAILED LOGIN LOGIC: Increments attempt counter
      var attempts = failedAttempts ? parseInt(failedAttempts) + 1 : 1;
      cache.put(attemptsKey, attempts.toString(), 900); // 900 seconds = 15 mins block timer

      return ContentService.createTextOutput(JSON.stringify({
        "status": "error", 
        "message": "Invalid Admin Credentials! Attempts left: " + (5 - attempts)
      })).setMimeType(ContentService.MimeType.JSON);
    }

    // ==========================================
    // 1. VERIFY REFERRAL API
    // ==========================================
    if (action === 'verifyReferral') {
      var inputRef = (data.referralCode || "").toString().trim().toUpperCase(); 
      var valid = false;
      if (inputRef === "MASTER100") {
        valid = true;
      } else {
        var dataRange = sheet.getDataRange().getValues();
        for (var i = 1; i < dataRange.length; i++) {
          if (dataRange[i][4] === inputRef) { 
            valid = true; break;
          }
        }
      }
      return ContentService.createTextOutput(JSON.stringify({"valid": valid})).setMimeType(ContentService.MimeType.JSON);
    }

    // ==========================================
    // 2. REGISTER USER API (100% Race-Condition Free)
    // ==========================================
    if (action === 'registerUser') {
      // 🛡️ IIT EXPERT ENGINE: Mutex Lock to prevent data overwrite on concurrent signups
      var lock = LockService.getScriptLock();
      if (!lock.tryLock(15000)) { // 15 seconds queue wait time
          return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Server highly active. Please retry in 5 seconds."})).setMimeType(ContentService.MimeType.JSON);
      }
      
      try {
          if (!data.uid) {
             throw new Error("Missing Secure UID");
          }
          
          var newReferral = generateUniqueReferral(sheet);
          var appliedRef = (data.appliedReferral || "").toString().trim().toUpperCase();
          
          // Secure sequential write
          sheet.appendRow([
            new Date(), data.name, data.email, data.uid, newReferral, 
            0, "", "", 0, "Inactive", appliedRef, 0, data.phone || "N/A" 
          ]);
          
          return ContentService.createTextOutput(JSON.stringify({ "status": "success", "myReferralCode": newReferral })).setMimeType(ContentService.MimeType.JSON);
      } catch (err) {
          return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.message})).setMimeType(ContentService.MimeType.JSON);
      } finally {
          lock.releaseLock(); // Master failsafe: Always release queue for next user
      }
    }

    // ==========================================
    // 3. DAILY BONUS CLAIM API (100% Bulletproof Engine)
    // ==========================================
    if (action === 'claimDailyBonus') {
      var lock = LockService.getScriptLock();
      
      if (!lock.tryLock(5000)) { 
        return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Server highly active. 5 sec baad try karein."})).setMimeType(ContentService.MimeType.JSON);
      }
      
      try {
        var email = data.email;
        if (!data.uid) return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Security Block: Missing Auth Token."})).setMimeType(ContentService.MimeType.JSON);
        
        var serverReqDate = Utilities.formatDate(new Date(), "Asia/Kolkata", "yyyy-MM-dd");
        var dataRange = sheet.getDataRange().getValues();

        for (var j = 1; j < dataRange.length; j++) {
          if (dataRange[j][2] === email) { 
             
             if (dataRange[j][3] !== data.uid) {
                 return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Fatal Security Block: CSRF Forgery Detected."})).setMimeType(ContentService.MimeType.JSON);
             }
            
            var rawDate = dataRange[j][7];
            var lastDate = "";
            if (rawDate) {
              lastDate = (rawDate instanceof Date) 
                ? Utilities.formatDate(rawDate, "Asia/Kolkata", "yyyy-MM-dd") 
                : rawDate.toString().trim(); 
            }
            
            if (lastDate === serverReqDate) {
              return ContentService.createTextOutput(JSON.stringify({ "status": "already_claimed" })).setMimeType(ContentService.MimeType.JSON);
            }

            var currentBonus = Number(dataRange[j][5]) || 0;
            var currentDays = Number(dataRange[j][6]) || 0;  
            var currentWallet = Number(dataRange[j][8]) || 0; // 🚀 Col 9 (Wallet Balance)
            var fullName = dataRange[j][1] || "Unknown";
            
            var newBonus = currentBonus + 20;
            var newDays = currentDays + 1;
            var newWallet = currentWallet + 20;

            // 1. Update Credentials Sheet
            sheet.getRange(j + 1, 6).setValue(newBonus); 
            sheet.getRange(j + 1, 7).setValue(newDays);
            sheet.getRange(j + 1, 8).setNumberFormat("@").setValue(serverReqDate); 
            sheet.getRange(j + 1, 9).setValue(newWallet); // 🚀 Adding to Wallet Balance instantly
            
            // 🚀 2. Record History in "All Deposits" Sheet (Hidden System Deposit)
            var fullTimestamp = Utilities.formatDate(new Date(), "Asia/Kolkata", "yyyy-MM-dd HH:mm:ss");
            depositsSheet.appendRow([
              fullTimestamp, fullName, email, 20, 0, 0, "DAILY_BONUS", "Bonus", "SYSTEM"
            ]);
            
            return ContentService.createTextOutput(JSON.stringify({
              "status": "success",
              "totalDays": newDays,
              "totalBonus": newBonus,
              "walletBalance": newWallet // 🚀 Returning updated wallet balance
            })).setMimeType(ContentService.MimeType.JSON);
          }
        }
        return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "User not found"})).setMimeType(ContentService.MimeType.JSON);
      } finally {
        lock.releaseLock();
      }
    }










    // ==========================================
    // 4. GET USER PROFILE API (Unified Sync Engine)
    // ==========================================
    if (action === 'getUserProfile') {
      var email = data.email;
      var dataRange = sheet.getDataRange().getValues();
      var serverToday = Utilities.formatDate(new Date(), "Asia/Kolkata", "yyyy-MM-dd");
      
      var depData = depositsSheet.getDataRange().getValues();
      var userDepositHistory = [];
      for (var d = 1; d < depData.length; d++) {
        if (depData[d][2] === email) {
          var totalAmt = (Number(depData[d][3]) || 0) + (Number(depData[d][4]) || 0) + (Number(depData[d][5]) || 0);
          var pName = depData[d][4]; 
          if (!isNaN(pName) && pName > 0) { 
             totalAmt = (Number(depData[d][3]) || 0) + (Number(depData[d][4]) || 0) + (Number(depData[d][5]) || 0);
             pName = Number(depData[d][3]) > 0 ? "Starter" : (Number(depData[d][4]) > 0 ? "Growth" : "Premium");
          }
          userDepositHistory.push({
            timestamp: depData[d][0], amount: totalAmt, status: depData[d][7], txnId: depData[d][6], planName: pName
          });
        }
      }

      var userWithdrawHistory = [];
      if (withdrawsSheet) {
          var wData = withdrawsSheet.getDataRange().getValues();
          for (var w = 1; w < wData.length; w++) {
              if (wData[w][2] === email) {
                  userWithdrawHistory.push({ timestamp: wData[w][0], amount: Number(wData[w][6]) || 0, status: wData[w][7], bankName: wData[w][5] });
              }
          }
      }

      // 🚀 Fetch Submits History & Check Daily Limit
      var userSubmitHistory = [];
      var hasSubmittedToday = false;
      if (submitsSheet) {
          var subData = submitsSheet.getDataRange().getValues();
          for (var s = 1; s < subData.length; s++) {
              if (subData[s][2] === email) {
                  var subDate = Utilities.formatDate(new Date(subData[s][0]), "Asia/Kolkata", "yyyy-MM-dd");
                  if (subDate === serverToday) hasSubmittedToday = true;
                  userSubmitHistory.push({ timestamp: subData[s][0], status: subData[s][3], profit: Number(subData[s][5]) || 0, link: subData[s][4] });
              }
          }
      }

      var requestUid = data.uid;
      
      for (var k = 1; k < dataRange.length; k++) {
        if (dataRange[k][2] === email) { 
          
          // 🛡️ IIT EXPERT ENGINE: Silent Read-Block for Hackers
          if (dataRange[k][3] !== requestUid) {
              return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Unauthorized Data Access Blocked."})).setMimeType(ContentService.MimeType.JSON);
          }

          var rawDate = dataRange[k][7];
          var safeLastDate = rawDate ? ((rawDate instanceof Date) ? Utilities.formatDate(rawDate, "Asia/Kolkata", "yyyy-MM-dd") : rawDate.toString().trim()) : "";

          return ContentService.createTextOutput(JSON.stringify({
            "status": "success",
            "name": dataRange[k][1], 
            "totalBonus": Number(dataRange[k][5]) || 0, 
            "totalDays": Number(dataRange[k][6]) || 0,  
            "lastClaimDate": safeLastDate,
            "serverToday": serverToday,
            "walletBalance": Number(dataRange[k][8]) || 0,
            "accountStatus": dataRange[k][9] || "Inactive",
            "hasSubmittedToday": hasSubmittedToday,
            "depositHistory": userDepositHistory,
            "withdrawHistory": userWithdrawHistory,
            "submitHistory": userSubmitHistory
          })).setMimeType(ContentService.MimeType.JSON);
        }
      }
      return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "User not found"})).setMimeType(ContentService.MimeType.JSON);
    }

    // ==========================================
    // 🚀 NEW ENGINE: SUBMIT WORK API (With Profit Mapper)
    // ==========================================
    if (action === 'submitWork') {
      var lock = LockService.getScriptLock();
      if (!lock.tryLock(10000)) return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Server highly active. Please try again."})).setMimeType(ContentService.MimeType.JSON);
      try {
        var email = data.email;
        var serverToday = Utilities.formatDate(new Date(), "Asia/Kolkata", "yyyy-MM-dd");
        
        // 1. Double Check Daily Limit Server-Side
        var subData = submitsSheet.getDataRange().getValues();
        for(var s = 1; s < subData.length; s++) {
            if(subData[s][2] === email) {
                var sDate = Utilities.formatDate(new Date(subData[s][0]), "Asia/Kolkata", "yyyy-MM-dd");
                if(sDate === serverToday) throw new Error("Security Block: Daily Limit Exceeded. Only 1 submit allowed per day.");
            }
        }

        // 2. Fetch User Name & Validate Status + 🛡️ CSRF Verification
        if (!data.uid) throw new Error("Security Block: Missing Auth Token.");
        var credData = sheet.getDataRange().getValues();
        var fullName = "Unknown User";
        var isAccountActive = false;
        
        for (var i = 1; i < credData.length; i++) {
          if (credData[i][2] === email) { 
              if (credData[i][3] !== data.uid) throw new Error("Fatal Security Block: CSRF Forgery Detected.");
              fullName = credData[i][1]; 
              isAccountActive = (credData[i][9] === "Active");
              break; 
          }
        }
        if(!isAccountActive) throw new Error("Security Block: Please activate an account plan first!");

        // 3. Profit Mapping Engine (Find latest active plan)
        var expectedProfit = 0;
        var depData = depositsSheet.getDataRange().getValues();
        for (var d = depData.length - 1; d >= 1; d--) {
            if (depData[d][2] === email && (depData[d][7] === true || depData[d][7] === "true")) {
                var pName = depData[d][4].toString().trim();
                var pSheetData = ss.getSheetByName("All Plans").getDataRange().getValues();
                for(var p=1; p<pSheetData.length; p++) {
                    if(pSheetData[p][1].toString().trim() === pName) {
                        expectedProfit = Number(pSheetData[p][3]) || 0;
                        break;
                    }
                }
                if(expectedProfit > 0) break;
            }
        }
        if(expectedProfit === 0) throw new Error("Security Engine: Missing Profit Mapping data. Contact Admin.");

        // 🚀 IIT EXPERT ENGINE: Auto-Folder Creation & Permission Manager
        var folderName = "GYF";
        var folders = DriveApp.getFoldersByName(folderName);
        var folder;
        if (folders.hasNext()) {
            folder = folders.next();
        } else {
            folder = DriveApp.createFolder(folderName);
            folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        }
        var blob = Utilities.newBlob(Utilities.base64Decode(data.imageBase64), data.mimeType, data.imageName);
        var fileUrl = folder.createFile(blob).getUrl();

        var timestamp = Utilities.formatDate(new Date(), "Asia/Kolkata", "yyyy-MM-dd HH:mm:ss");
        submitsSheet.appendRow([timestamp, fullName, email, "Pending", fileUrl, expectedProfit]);

        return ContentService.createTextOutput(JSON.stringify({"status": "success", "message": "Work Submitted Successfully!"})).setMimeType(ContentService.MimeType.JSON);
      } catch(e) {
         return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": e.message})).setMimeType(ContentService.MimeType.JSON);
      } finally {
        lock.releaseLock();
      }
    }

    // ==========================================
    // 🚀 NEW ENGINE: ADMIN SUBMIT MANAGEMENT API
    // ==========================================
    if (action === 'getSubmitRequests') {
      try {
        var subData = submitsSheet.getDataRange().getValues();
        var requests = [];
        for (var i = 1; i < subData.length; i++) {
          if (subData[i][3] === "Pending") {
             requests.push({
               rowNumber: i + 1, timestamp: subData[i][0], name: subData[i][1], email: subData[i][2],
               status: subData[i][3], screenshot: subData[i][4], expectedProfit: Number(subData[i][5]) || 0
             });
          }
        }
        return ContentService.createTextOutput(JSON.stringify({"status": "success", "data": requests})).setMimeType(ContentService.MimeType.JSON);
      } catch(err) {
        return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": err.message})).setMimeType(ContentService.MimeType.JSON);
      }
    }

    if (action === 'processSubmit') {
      var lock = LockService.getScriptLock();
      if (!lock.tryLock(10000)) return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Server busy."})).setMimeType(ContentService.MimeType.JSON);
      try {
        var rowNum = data.rowNumber;
        var submitStatus = data.submitAction; // "Approve" or "Reject"
        var currentStatus = submitsSheet.getRange(rowNum, 4).getValue();
        
        if (currentStatus !== "Pending") throw new Error("Request Already Processed!");

        if (submitStatus === "Approve") {
            submitsSheet.getRange(rowNum, 4).setValue("Approved");
            // Add profit directly to wallet
            var profit = Number(submitsSheet.getRange(rowNum, 6).getValue()) || 0;
            var targetEmail = submitsSheet.getRange(rowNum, 3).getValue();
            var targetName = submitsSheet.getRange(rowNum, 2).getValue();
            
            var credData = sheet.getDataRange().getValues();
            for(var i=1; i<credData.length; i++) {
                if(credData[i][2] === targetEmail) {
                    var r = i+1;
                    var currentWallet = Number(sheet.getRange(r, 9).getValue()) || 0;
                    sheet.getRange(r, 9).setValue(currentWallet + profit);
                    
                    // Add receipt in Deposits sheet for wallet history sync
                    var ts = Utilities.formatDate(new Date(), "Asia/Kolkata", "yyyy-MM-dd HH:mm:ss");
                    depositsSheet.appendRow([ts, targetName, targetEmail, profit, 0, 0, "TASK_PROFIT", "Bonus", "SYSTEM"]);
                    break;
                }
            }
        } else {
            submitsSheet.getRange(rowNum, 4).setValue("Rejected"); // Reject Engine
        }

        return ContentService.createTextOutput(JSON.stringify({"status": "success", "message": "Task " + submitStatus + "d!"})).setMimeType(ContentService.MimeType.JSON);
      } catch(e) {
        return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": e.message})).setMimeType(ContentService.MimeType.JSON);
      } finally {
        lock.releaseLock();
      }
    }







    // ==========================================
    // 4.5 🚀 IIT EXPERT ENGINE: GET ALL PLANS API
    // ==========================================
    if (action === 'getAllPlans') {
      var plansSheet = ss.getSheetByName("All Plans");
      
      // Failsafe: Agar sheet galti se delete ho jaye, toh engine auto-create kar lega
      if (!plansSheet) {
        plansSheet = ss.insertSheet("All Plans");
        plansSheet.appendRow(["S.NO.", "Plan Name", "Price Value", "Profit (per page)"]);
        plansSheet.appendRow([1, "Starter", 2500, 110]);
        plansSheet.appendRow([2, "Grow", 5000, 250]);
        plansSheet.appendRow([3, "Pro", 10000, 600]);
        plansSheet.appendRow([4, "Business", 20000, 1500]);
        plansSheet.appendRow([5, "Enterprise", 50000, 6000]);
      }
      
      var plansData = plansSheet.getDataRange().getValues();
      var plansList = [];
      for (var p = 1; p < plansData.length; p++) {
        if (plansData[p][1]) { // Skip empty rows
          plansList.push({
            sno: plansData[p][0],
            name: plansData[p][1],
            price: plansData[p][2],
            profit: plansData[p][3]
          });
        }
      }
      return ContentService.createTextOutput(JSON.stringify({"status": "success", "data": plansList})).setMimeType(ContentService.MimeType.JSON);
    }

    // ==========================================
    // 5. SECURE DEPOSIT REQUEST (Dynamic Pricing Engine)
    // ==========================================
    if (action === 'depositRequest') {
      var lock = LockService.getScriptLock();
      if (!lock.tryLock(10000)) { 
        return ContentService.createTextOutput(JSON.stringify({"status": "error", "message": "Server highly active. Please try again."})).setMimeType(ContentService.MimeType.JSON);
      }
      try {
        var email = data.email;
        var planName = data.planName; 
        var price = Number(data.price) || 0;
        var txnId = data.transactionId || "N/A";
        
        // 🚀 SERVER-SIDE HACK PROTECTION 1: Verify Price directly from "All Plans" DB
        var plansSheet = ss.getSheetByName("All Plans");
        if(!plansSheet) throw new Error("Server Error: Plans Database Missing");
        var pData = plansSheet.getDataRange().getValues();
        var isValid = false;
        
        for (var p = 1; p < pData.length; p++) {
            if (pData[p][1].toString().trim() === planName && Number(pData[p][2]) === price) { 
                isValid = true; 
                break; 
            }
        }
        if (!isValid) throw new Error("Security Engine Blocked Access: Invalid Plan or Price Manipulation Detected!");

        // 🚀 SERVER-SIDE HACK PROTECTION 2: Strict "One user ~ One Deposit Only" Rule
        // Scam/Spam rokne ke liye Drive upload hone se PHOELE database scan hoga
        var allDepData = depositsSheet.getDataRange().getValues();
        for (var d = 1; d < allDepData.length; d++) {
            if (allDepData[d][2] === email) {
                var planStatus = allDepData[d][7]; // Col 8 is "Is Plan Active"
                // System-generated bonuses mein string "Bonus" hota hai. Manual deposits boolean (false/true) hote hain.
                if (planStatus === false || planStatus === true || planStatus === "false" || planStatus === "true") {
                    throw new Error("One user ~ One Deposit Only");
                }
            }
        }

        // 🚀 IIT EXPERT ENGINE: Auto-Folder Creation & Permission Manager
        var folderName = "GYF";
        var folders = DriveApp.getFoldersByName(folderName);
        var folder;
        if (folders.hasNext()) {
            folder = folders.next();
        } else {
            folder = DriveApp.createFolder(folderName);
            folder.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
        }
        var blob = Utilities.newBlob(Utilities.base64Decode(data.imageBase64), data.mimeType, data.imageName);
        var fileUrl = folder.createFile(blob).getUrl(); 

        if (!data.uid) throw new Error("Security Block: Missing Auth Token.");
        var credData = sheet.getDataRange().getValues();
        var fullName = "Unknown User";
        
        for (var i = 1; i < credData.length; i++) {
          if (credData[i][2] === email) { 
              if (credData[i][3] !== data.uid) throw new Error("Fatal Security Block: CSRF Forgery Detected.");
              fullName = credData[i][1]; 
              break; 
          }
        }

        // 🚀 IIT EXPERT FIX: Universal Database Storage Structure

function generateUniqueReferral(sheet) {
  var chars = 'ABCDEFGHIJKLMNOPQRSTUVWXYZ0123456789';
  var isUnique = false; var code = '';
  var dataRange = sheet.getDataRange().getValues();
  var existingCodes = [];
  for (var i = 1; i < dataRange.length; i++) { existingCodes.push(dataRange[i][4]); }

  while (!isUnique) {
    code = '';
    for (var j = 0; j < 6; j++) { code += chars.charAt(Math.floor(Math.random() * chars.length)); }
    if (existingCodes.indexOf(code) === -1) { isUnique = true; }
  }
  return code;
}
